public class DIP {

    interface Interface {
        void method();
    }

    class ClassI implements Interface{ // OK
        Interface i;

        public void method() {};

        ClassI(Interface i) {
            this.i = i; // OK
        }

        ClassI() {
            this.i = new Interface() {public void method() {} }; // Violation DIP#1
        }
    }

    abstract class ABS {
        abstract void method(Interface i); // OK
        abstract void method(ABS a); // OK
        abstract void method(ClassI c); // Violation DIP#2
    }

    class ClassImpl extends ABS { // OK
        void method(Interface i) {}
        void method(ABS a) {}
        void method(ClassI c) {}
    }

    class ClassIsOK7 extends ClassImpl {} // OK, it is just an inheritance ?
}
